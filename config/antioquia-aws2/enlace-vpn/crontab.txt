# Backup diario a las 02:00 AM
# Genera volcado completo de la BD damas en /home/ubuntu/backups/
# y elimina backups con más de 7 días
0 2 * * * /home/ubuntu/backup_damas.sh >> /var/log/backup_damas.log 2>&1

# Limpieza de partidas expiradas — cada hora
# Llama al procedimiento limpiar_expiradas() que elimina
# partidas con expira_en < NOW() para mantener la BD limpia
0 * * * * sudo mysql damas -e 'CALL limpiar_expiradas();' >> /var/log/damas_cron.log 2>&1
